    1. Create webfrontend/src/components/UsageDashboard.jsx:
       ```jsx
       import React, { useState, useEffect } from 'react';
       import axios from 'axios';
       
       export function UsageDashboard() {
         const [balance, setBalance] = useState(null);
         const [loading, setLoading] = useState(true);
         const [error, setError] = useState(null);
         
         useEffect(() => {
           axios.get('/api/balance_list/')
             .then(res => {
               setBalance(res.data.balances[0] || { usage: 0, limit: 100 });
               setLoading(false);
             })
             .catch(err => {
               setError('Failed to load balance');
               setLoading(false);
             });
         }, []);
         
         if (loading) return <div>Loading...</div>;
         if (error) return <div className="error">{error}</div>;
         
         const percentage = (balance.usage / balance.limit) * 100;
         const warningThreshold = 80;
         const isWarning = percentage >= warningThreshold;
         
         // Color based on usage
         const getColor = () => {
           if (percentage < 50) return 'bg-green-500';
           if (percentage < 80) return 'bg-yellow-500';
           return 'bg-red-500';
         };
         
         return (
           <div className="usage-dashboard p-6 bg-white rounded-lg shadow">
             <h2 className="text-xl font-bold mb-4">Usage Overview</h2>
             
             <div className="mb-4">
               <div className="flex justify-between mb-2">
                 <span>Usage: {balance.usage}</span>
                 <span>Limit: {balance.limit}</span>
               </div>
               <div className="w-full bg-gray-200 rounded-full h-4">
                 <div 
                   className={`h-4 rounded-full ${getColor()} transition-all`}
                   style={{ width: `${Math.min(percentage, 100)}%` }}
                 />
               </div>
             </div>
             
             {isWarning && (
               <div className="bg-red-100 border border-red-400 text-red-700 px-4 py-2 rounded">
                 ⚠️ Warning: You've used {percentage.toFixed(0)}% of your limit!
               </div>
             )}
           </div>
         );
       }
       ```
    
    2. Use Tailwind CSS for styling (assumes Tailwind is available)
    
    3. Include loading and error states
    
    4. Add smooth transitions for progress bar
